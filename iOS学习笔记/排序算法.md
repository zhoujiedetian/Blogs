##### 快速排序

```
func quickSort(_ arr: inout [Int], _ leftIndex: Int, _ rightIndex: Int) {
    if leftIndex > rightIndex {
        return
    }
    var i = leftIndex
    var j = rightIndex
    while i < j {
        while i < j && arr[j] >= arr[leftIndex] {
            j -= 1
        }
        while i < j && arr[i] <= arr[leftIndex] {
            i += 1
        }
        let tmp = arr[i]
        arr[i] = arr[j]
        arr[j] = tmp
    }
    let tmp = arr[leftIndex]
    arr[leftIndex] = arr[i]
    arr[i] = tmp
    quickSort(&arr, leftIndex, i - 1)
    quickSort(&arr, i + 1, rightIndex)
}
```

```
/**
 冒泡排序,最坏O(n^2),最好O(n)，平均O(n^2)
 */
func sort1(_ arr: [Int]) -> [Int] {
    var result = arr
    let n = arr.count
    for i in 0..<n {
        for j in i+1..<n {
            if result[j] < result[i] {
                let tmp = result[i]
                result[i] = result[j]
                result[j] = tmp
            }
        }
    }
    return result
}
```

```
func sort2(_ arr: [Int]) -> [Int] {
    var result = arr
    let n = arr.count
    for i in 0..<n {
        var minIndex = i
        for j in i+1..<n {
            if result[j] < result[minIndex] {
                minIndex = j
            }
        }
        let tmp = result[i]
        result[i] = result[minIndex]
        result[minIndex] = tmp
    }
    return result
}
```

```
/**
 插入排序,最坏O(n^2),最好O(n)，平均O(n^2)
 */
func sort3(_ arr: [Int]) -> [Int] {
    var result = arr
    let n = arr.count
    for i in 1..<n {
        var j = i
        while j > 0 {
            if result[j-1] > result[j] {
                let tmp = result[j-1]
                result[j-1] = result[j]
                result[j] = tmp
                j -= 1
            }else {
                break
            }
        }
    }
    return result
}
```

##### 希尔排序

```
/**
 希尔排序，最坏O(n^2),最好O(n)，平均O(n^1.3)
 */
func sort4(_ arr: [Int]) -> [Int] {
    var result = arr
    let n = arr.count
    var step = n/2
    while step > 0 {
        for i in step..<n {
            var fromIndex = i - step
            while fromIndex >= 0 {
                if result[fromIndex] > result[fromIndex + step] {
                    let tmp = result[fromIndex]
                    result[fromIndex] = result[fromIndex + step]
                    result[fromIndex+step] = tmp
                }
                fromIndex -= step
            }
        }
        step = step / 2
    }
    return result
}
```

```
/**

 */
func sort5(_ arr: [Int]) -> [Int] {
    var result = [Int]()
    var buckets = [[Int]](repeating: [Int](), count: 11)
    for num in arr {
        buckets[num].append(num)
    }
    for bucket in buckets {
        for num in bucket {
            result.append(num)
        }
    }
    return result
}
```

```
/**
 桶排序
 */
func sort6(_ arr: [Int]) -> [Int] {
    var result = [Int]()
    var minVal = arr[0]
    var maxVal = arr[0]
    for num in arr {
        if num < minVal {
            // 最小数
            minVal = num
        }else if num > maxVal {
            // 最大数
            maxVal = num
        }
    }
    var buckets = [Int](repeating: 0, count: maxVal - minVal + 1)
    for num in arr {
        var index = buckets[num-minVal]
        index += 1
        buckets[num-minVal] = index
    }
    for i in 0..<buckets.count {
        if buckets[i] > 0 {
            for _ in 0..<buckets[i] {
                result.append(i+minVal)
            }
        }
    }
    return result
}
```
